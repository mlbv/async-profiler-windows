# Shipped binaries are stripped: their CodeView debug information goes to a PDB that they name without its
# build path, as upstream moves its library's debug information into a separate -debug package.
function(separate_debug_symbols target)
    target_compile_options(${target} PRIVATE -gcodeview)
    target_link_options(${target} PRIVATE -s LINKER:--pdb= LINKER:-Xlink=-pdbaltpath:%_PDB%)
endfunction()
